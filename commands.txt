# ============================================================
#  汉字笔顺 MCP 项目 - 常用命令
# ============================================================

# 环境要求
#   - Node.js >= 20.0.0
#   - npm 或 pnpm

# ------------------------------------------------------------
#  1. 安装依赖
# ------------------------------------------------------------
npm install

# ------------------------------------------------------------
#  2. 开发模式（带热更新）
# ------------------------------------------------------------
npm run dev

# ------------------------------------------------------------
#  3. 类型检查
# ------------------------------------------------------------
npm run typecheck

# ------------------------------------------------------------
#  4. 生产构建
# ------------------------------------------------------------
npm run build

# ------------------------------------------------------------
#  5. 启动生产服务
# ------------------------------------------------------------
npm start

# ------------------------------------------------------------
#  6. 运行测试
# ------------------------------------------------------------
npm test

# CI 环境下一次性运行测试
npm run test:ci

# ------------------------------------------------------------
#  7. 清理构建产物
# ------------------------------------------------------------
npm run clean

# ------------------------------------------------------------
#  8. Web 预览（Next.js 预览 MCP UI）
# ------------------------------------------------------------
cd web
npm install
npm run dev
# 然后访问 http://localhost:3000

# ------------------------------------------------------------
#  9. Docker 部署
# ------------------------------------------------------------
# 构建镜像
docker build -t hanzi-write-mcp .

# 运行容器
docker run -p 3000:3000 hanzi-write-mcp

# ------------------------------------------------------------
#  10. 环境变量
# ------------------------------------------------------------
# 复制 .env.example 为 .env 并按需修改
cp .env.example .env
